\section{Kết quả thực nghiệm}
\label{sec:results}

Phần này trình bày kết quả huấn luyện và đánh giá của 5 mô hình trên cả hai tập dữ liệu Lúa và Cà phê. Tất cả mô hình được huấn luyện với cùng thiết lập chung (Mục~\ref{sec:training_config}) nhằm đảm bảo so sánh công bằng.

% ============================================================

\label{subsec:learning_curves}

Hình~\ref{fig:lc_rice} và Hình~\ref{fig:lc_coffee} trình bày đường cong học (training loss và validation loss) của 5 mô hình qua 10 epoch trên từng tập dữ liệu.

\begin{figure}[H]
    \centering
    \includegraphics[width=0.95\textwidth]{img/learning_curves_coffee.png}
    \caption{Learning curves của 5 mô hình trên tập Cà phê. Đường liền: validation loss; đường đứt: training loss.}
    \label{fig:lc_coffee}
\end{figure}

\textbf{Nhận xét:}
\begin{itemize}
    \item \textbf{RF-DETR} và \textbf{YOLO26m-seg} hội tụ nhanh nhất trên cả hai tập dữ liệu, đạt validation loss thấp nhất trong số 5 mô hình. Hai mô hình này có khoảng cách train--val loss nhỏ, cho thấy khả năng tổng quát hóa tốt và không xảy ra overfitting nghiêm trọng trong 10 epoch.
    \item \textbf{Mask2Former} hội tụ ổn định, validation loss nằm ở mức trung bình; khoảng cách train--val nhỏ cho thấy mô hình không bị overfitting.
    \item \textbf{Mask R-CNN} có validation loss cao hơn, đặc biệt trên tập Rice, phản ánh hạn chế của kiến trúc hai giai đoạn với kích thước ảnh nhỏ ($256 \times 256$).
    \item \textbf{MobileSAM} có validation loss cao nhất do chỉ fine-tune Mask Decoder ($\approx 3,87$M tham số), hạn chế khả năng học đặc trưng mới. Tuy nhiên, khoảng cách train--val rất nhỏ nhờ đóng băng encoder, hoàn toàn tránh được overfitting.
    \item Trên tập \textbf{Coffee}, tất cả mô hình đều đạt validation loss thấp hơn so với Rice, cho thấy các vết bệnh trên lá cà phê có đặc trưng thị giác rõ ràng và dễ học hơn.
\end{itemize}

\textbf{Nhận xét và phân tích:}

\begin{itemize}
    \item \textbf{RF-DETR đạt hiệu năng cao nhất} trên tất cả các metric ở cả hai tập dữ liệu. Trên Coffee, mAP@50:95 đạt $0,897$ và Dice đạt $0,935$; trên Rice, mAP@50:95 đạt $0,826$ và Dice đạt $0,885$. Cơ chế attention toàn cục của Transformer giúp mô hình nắm bắt tốt quan hệ ngữ cảnh giữa vùng bệnh và nền lá.

    \item \textbf{YOLO26m-seg có hiệu năng rất sát RF-DETR}, chỉ thấp hơn khoảng 2--3 đơn vị phần trăm trên các metric chính (mAP@50:95 Rice: $0,801$ vs $0,826$; Coffee: $0,871$ vs $0,897$). Tuy nhiên, \textbf{YOLO26m-seg có tốc độ suy luận nhanh nhất}, gấp $\sim$5,5× so với RF-DETR ($12,3$ ms vs $68,5$ ms trên Rice) --- đây là lợi thế quyết định khi triển khai ứng dụng thời gian thực.

    \item \textbf{Mask2Former} đạt kết quả ổn định ở mức trung bình--khá, mAP@50:95 đạt $0,731$ (Rice) và $0,828$ (Coffee), với tốc độ suy luận $\sim$90 ms.

    \item \textbf{Mask R-CNN} (baseline CNN) có hiệu năng thấp hơn các mô hình Transformer, đặc biệt trên Rice ($0,653$ mAP@50:95). Kích thước ảnh nhỏ ($256 \times 256$) hạn chế khả năng phát hiện vết bệnh nhỏ. Tốc độ suy luận chậm ($\sim$280 ms) do kiến trúc hai giai đoạn.

    \item \textbf{MobileSAM} có hiệu năng thấp nhất do chỉ fine-tune Mask Decoder. Tuy nhiên, mô hình vẫn đạt mIoU $> 0,62$ nhờ thừa hưởng khả năng trích xuất đặc trưng tổng quát từ pretrained TinyViT. Tốc độ suy luận rất chậm ($> 1400$ ms) do kiến trúc nặng của SAM.

    \item Trên cả hai tập dữ liệu, \textbf{Coffee luôn cho kết quả tốt hơn Rice}. Nguyên nhân: vết bệnh trên lá cà phê (gỉ sắt, đốm rong) thường có đặc trưng màu sắc tương phản mạnh và ranh giới rõ ràng hơn so với vết bệnh nhỏ, phân tán trên lá lúa.
\end{itemize}

\subsection{Phân tích lỗi và các trường hợp dự đoán sai}
\label{subsec:error_analysis}

Để hiểu rõ hơn về các hạn chế hiện tại của mô hình, nhóm tiến hành phân tích định tính các mẫu dữ liệu dự đoán sai tiêu biểu của mô hình baseline Mask R-CNN trên tập Test (được trực quan hóa trong Hình~\ref{fig:error_analysis}).

\begin{figure}[H]
    \centering
    \includegraphics[width=0.85\textwidth]{img/analysis_image.png}
    \caption{Các ví dụ trực quan về dự đoán sai (ảnh gốc, nhãn thực tế Ground Truth và dự đoán của mô hình Mask R-CNN) trên tập dữ liệu Rice.}
    \label{fig:error_analysis}
\end{figure}

Dựa trên kết quả trực quan ở Hình~\ref{fig:error_analysis}, các lỗi sai của mô hình tập trung vào hai nhóm nguyên nhân chính:

\begin{enumerate}
    \item \textbf{Sự mập mờ về ranh giới đặc trưng thị giác ở mức vi mô (Ambiguous micro-features):}
          \begin{itemize}
              \item \textit{Mô tả lỗi:} Ở các mẫu lá lúa \textbf{Image 577}, \textbf{Image 579} và \textbf{Image 580}, Ground Truth dán nhãn cho toàn bộ chiếc lá là lớp \textbf{Healthy} (được biểu thị bằng màu xanh lá). Tuy nhiên, mô hình Mask R-CNN lại dự đoán nhãn \textbf{BrownSpot} (Bệnh đốm nâu) với độ tin cậy rất cao (lần lượt là $0.96$, $0.72$, $0.87$) và tô đỏ toàn bộ chiếc lá.
              \item \textit{Giả thuyết nguyên nhân:} Trên thực tế, trên các phiến lá Healthy này xuất hiện các vết đốm hoại tử rất nhỏ, các đốm xước cơ học li ti hoặc các đốm vàng do thiếu chất dinh dưỡng nhẹ. Những đốm nhỏ này có đặc trưng hình học (hình chấm tròn nhỏ) và màu sắc (nâu sẫm/vàng nhạt) rất tương đồng với triệu chứng ban đầu của bệnh đốm nâu. Do đó, mô hình baseline dễ bị nhầm lẫn bởi các "nhiễu vi mô" này và phóng đại kết quả phân loại thành bệnh đốm nâu cho toàn bộ chiếc lá. Ranh giới đặc trưng giữa "lá khỏe mạnh có tì vết nhỏ" và "lá bị bệnh đốm nâu giai đoạn chớm nở" là cực kỳ mong manh, khiến mô hình gặp khó khăn lớn trong việc phân biệt.
          \end{itemize}

    \item \textbf{Lỗi định vị vùng (Localization/Segmentation leakage) do nhiễu nền:}
          \begin{itemize}
              \item \textit{Mô tả lỗi:} Ở mẫu \textbf{Image 578}, ảnh gốc chứa một chiếc lá lúa xanh nằm dọc ở mép trái với phần nền trắng. Nhãn Ground Truth khoanh vùng chính xác chiếc lá này. Tuy nhiên, mô hình dự đoán nhãn \textbf{BrownSpot} ($0.84$) và \textbf{LeafBlast} ($0.58$) đồng thời tạo ra một mặt nạ (mask) khổng lồ màu đỏ bao phủ gần như toàn bộ bức ảnh bao gồm cả nền trống.
              \item \textit{Giả thuyết nguyên nhân:} Đây là một lỗi nghiêm trọng liên quan đến nhánh đề xuất vùng (RPN - Region Proposal Network) và cơ chế RoI Align của Mask R-CNN. Khi độ tương phản của biên vật thể so với nền không đủ mạnh hoặc do sự phân tán đặc trưng từ các bộ lọc tích chập, mô hình đã đề xuất một vùng anchor quá lớn (bao phủ cả nền) và không thể giới hạn mặt nạ bên trong biên chiếc lá. Lỗi này phản ánh điểm hạn chế của kiến trúc CNN hai giai đoạn khi xử lý các vật thể dài, mảnh trên nền sáng, dễ bị ảnh hưởng bởi sự lan truyền lỗi (error propagation) từ giai đoạn đề xuất vùng sang giai đoạn phân đoạn pixel.
          \end{itemize}
\end{enumerate}
