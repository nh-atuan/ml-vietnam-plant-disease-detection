\textbf{Gợi ý phân tích khi điền:}
\begin{itemize}
    \item \textbf{Overfitting:} Train mAP cao nhưng Val mAP thấp hơn đáng kể ($>10\%$); Loss train tiếp tục giảm nhưng loss val bắt đầu tăng sau một số epoch.
    \item \textbf{Underfitting:} Cả train và val đều thấp; loss không giảm hoặc giảm rất chậm.
    \item \textbf{Hội tụ tốt:} Train và val loss đều giảm ổn định; gap nhỏ và ổn định.
\end{itemize}

\textbf{Nguyên nhân sai dự đoán (giả thuyết):}
\begin{enumerate}
    \item \textbf{Nhiễu nền thực địa:} Ảnh chụp ngoài đồng có nền phức tạp (đất, cỏ, ánh sáng không đều), mô hình nhầm vùng nền là bệnh.
    \item \textbf{Vết bệnh quá nhỏ:} Các vết bệnh giai đoạn đầu kích thước $< 10\times10$ px --- khó phát hiện ở độ phân giải $256\times256$.
    \item \textbf{Chồng lấp lá:} Nhiều lá chồng nhau làm mô hình khó tách biệt instance.
    \item \textbf{Phân phối lớp mất cân bằng:} Lớp \texttt{Healthy} chiếm tỷ lệ cao, mô hình có xu hướng predict healthy nhiều hơn.
\end{enumerate}
