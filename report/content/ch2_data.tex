\section{Thu thập và Phân tích Dữ liệu}

\subsection{Nguồn và Phương pháp Thu thập}
\begin{itemize}
  \item Trình bày nguồn dữ liệu (API, scraping, tự tạo,\ldots).
  \item Mô tả chi tiết quy trình và công cụ thu thập, định dạng lưu trữ.
\end{itemize}

\subsection{Tiền xử lý và Làm sạch}
\begin{itemize}
  \item Mô tả các bước đã thực hiện: xử lý nhiễu, dữ liệu thiếu, chuẩn hóa, biến đổi.
\end{itemize}

\subsection{Phân tích Khám phá Dữ liệu (EDA)}
\begin{itemize}
  \item Trình bày các thống kê mô tả, phân tích phân bố nhãn, và mối quan hệ giữa các thuộc tính.
  \item Sử dụng biểu đồ, bảng để minh họa các phát hiện quan trọng về dữ liệu.
  \item Phân tích chất lượng dữ liệu và cách xử lý các vấn đề (ngoại lệ, trùng lặp).
\end{itemize}
