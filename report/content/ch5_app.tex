\section{Xây dựng và Triển khai Ứng dụng}

\subsection{Kiến trúc Hệ thống}
\begin{itemize}
  \item Mô tả kiến trúc tổng thể của ứng dụng, cách mô hình được đóng gói và tích hợp.
\end{itemize}

\subsection{Giao diện và Chức năng}
\begin{itemize}
  \item Mô tả giao diện người dùng (UI) và các chức năng chính.
  \item Cung cấp hình ảnh chụp màn hình ứng dụng.
\end{itemize}

\subsection{Triển khai}
\begin{itemize}
  \item Nêu rõ nền tảng triển khai (cloud công khai hoặc local).
  \item Nếu triển khai trên nền tảng cloud công khai: Cung cấp đường dẫn URL của ứng dụng đang hoạt động.
  \item Nếu triển khai local: Cung cấp hướng dẫn chi tiết (các bước cài đặt, câu lệnh chạy) để có thể chạy ứng dụng trên máy local.
\end{itemize}
