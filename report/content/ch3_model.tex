\section{Lựa chọn và Huấn luyện Mô hình}

\subsection{Chuẩn bị Dữ liệu cho Mô hình}
\begin{itemize}
  \item Nêu rõ tỷ lệ chia tập Train/Validation/Test và lý do.
  \item Tóm tắt các bước tiền xử lý cuối cùng trước khi đưa vào mô hình.
\end{itemize}

\subsection{Lựa chọn và Kiến trúc Mô hình}
\begin{itemize}
  \item Trình bày và so sánh ít nhất 3 mô hình đã thử nghiệm.
  \item Lý giải lý do lựa chọn. Với mô hình Deep Learning, vẽ sơ đồ và mô tả kiến trúc chi tiết.
\end{itemize}

\subsection{Cấu hình Huấn luyện}
\begin{itemize}
  \item Mô tả hàm mất mát, thuật toán tối ưu, và các siêu tham số chính.
  \item Trình bày phương pháp tinh chỉnh tham số (Grid Search, Random Search,\ldots).
\end{itemize}
