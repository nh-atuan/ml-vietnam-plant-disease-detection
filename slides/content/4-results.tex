\begin{frame}{Kết quả tinh chỉnh YOLO26-seg bằng RayTune và ASHA}
  \begin{table}
    \centering
    \caption{So sánh trước và sau tinh chỉnh siêu tham số}
    \small
    \resizebox{\textwidth}{!}{
      \begin{tabular}{llcccccc}
