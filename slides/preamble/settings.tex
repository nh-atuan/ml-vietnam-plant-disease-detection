%------------- Mục lục đầu mỗi section -------------
\newbool{showtocatsection}
\booltrue{showtocatsection}
\AtBeginSection[]{
  \ifbool{showtocatsection}{
    \begin{frame}{Mục lục}
      \tableofcontents[currentsection,subsectionstyle=hide]
    \end{frame}
  }{}
}

% Không chèn mục lục ở đầu mỗi subsection để deck không vượt quá 40 slide.

% Căn đều (justify) toàn bộ các khối văn bản trong slide
\apptocmd{\frame}{}{\justifying}{}
\addtobeamertemplate{block begin}{}{\justifying}
\makeatletter
\apptocmd{\beamer@column}{\justifying}{}{}

% Tự động thêm frametitle từ subsection nếu slide chưa có tiêu đề
\patchcmd{\endbeamer@frameslide}{\ifx\beamer@frametitle\@empty}{\iffalse}{}{\typeout{patch failed}}

\setbeamertemplate{frametitle}
{
  \ifx\insertframetitle\@empty
    \ifx\insertsubsection\@empty
      % Empty title and no subsection - draw nothing
    \else
      \nointerlineskip%
      \begin{beamercolorbox}[wd=\paperwidth,leftskip=0.3cm,rightskip=0.3cm,sep=0.3cm,dp=0.2ex]{frametitle}%
        \usebeamerfont{frametitle}\thesection.\thesubsection. \insertsubsection%
      \end{beamercolorbox}%
    \fi
  \else
    \nointerlineskip%
    \begin{beamercolorbox}[wd=\paperwidth,leftskip=0.3cm,rightskip=0.3cm,sep=0.3cm,dp=0.2ex]{frametitle}%
      \usebeamerfont{frametitle}\insertframetitle%
    \end{beamercolorbox}%
  \fi
}
\makeatother
